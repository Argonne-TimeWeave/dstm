% ---------------------------------------------------------------------------
%  Processes and ranks
%
%  A *process* is an entity: one running instance of the program. A *rank* is
%  an integer naming it. They are never identified.
%
%  Convention, used without exception:
%    UPPERCASE  P, Q, ...   range over processes
%    lowercase  p, q, ...   range over ranks
%
%  Every structure is indexed by rank, distances included; a process appears
%  only where an entity acts.
% ---------------------------------------------------------------------------
\newcommand{\worldsize}{n}                       % number of processes
\newcommand{\Ranks}{\mathcal{R}}                 % the rank set, {0,...,n-1}
\newcommand{\Procs}{\mathcal{P}}                 % the set of processes
\newcommand{\proc}[1]{P_{#1}}                    % the process of rank #1
\newcommand{\rankof}[1]{\mathrm{rank}(#1)}       % the rank of a process
\newcommand{\succr}[1]{#1 \oplus 1}              % ring successor of a rank

% ---------------------------------------------------------------------------
%  Communication graph and the induced distance, both on ranks
%
%    * the graph (R,E,w) says which ranks may exchange operations and what
%      each channel costs. E is sparse; DSTM exploits that.
%    * d is the extended metric E induces. It is total and is *not*
%      adjacency: two ranks may be at finite distance sharing no channel.
% ---------------------------------------------------------------------------
\newcommand{\cgraph}{\mathcal{G}}                % the communication graph
\newcommand{\edges}{E}                           % its edge set
\newcommand{\ew}[2]{w(#1,#2)}                    % channel weight on an edge
\newcommand{\dist}[2]{d(#1,#2)}                  % the induced distance
\newcommand{\Dmat}{\mathbf{D}}                   % API-level matrix encoding
\newcommand{\noedge}{\bot}                       % absence of an edge
\newcommand{\nbr}[1]{N(#1)}                      % neighbourhood (adjacency)
\newcommand{\diam}{\Delta}                       % max distance in a neighbourhood
